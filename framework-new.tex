\chapter{[DRAFT scaffold] The abstract framework}\label{ch.framework_DRAFT}
%============================================================================

This chapter takes a list of ingredients---a \emph{rewiring setup}---and
produces, with no further input, a syntax operad together with a functor into
$\org$. The smooth development of the main text ($\cat M=\mfd$, $\cat V=\rvect$,
$\Fun{c}=\cot$, $R=\rr$) is recovered by plugging in (\cref{thm.recovery}).
Throughout, the target categories $\poly$ and $\org$ are \emph{fixed}: every
setup feeds the same semantic anchor, which is what makes ``a category over
$\org$'' the natural output type.

%----------------------------------------------------------------------------
\section{Rewiring setups}\label{sec.rewiring_setup}
%----------------------------------------------------------------------------

\begin{definition}[Rewiring setup]\label{def.rewiring_setup}
A \emph{rewiring setup} $D$ consists of
\begin{enumerate}
  \item a cartesian monoidal category $\cat M$, the \emph{spaces};
  \item a symmetric monoidal category $\cat V$, the \emph{parameters}, together
    with a strong monoidal functor $J\colon\cat V\to\cat M$;
  \item a commutative monoid object $R$ in $\cat M$, the \emph{potential receptor};
  \item a strong symmetric monoidal functor
    $\Fun{c}\colon(\cat M,1,\times)\to(\poly,\yon,\otimes)$, the \emph{interface functor};
  \item a $\otimes$-monoid $z$ in $\poly$ equipped with a
    $\bigl(\Fun{c}(R)\otimes\blank\bigr)$-monoid structure $\alpha$
    (\cref{def.T_monoid}), the \emph{potential algebra}.
\end{enumerate}
\end{definition}

The five pieces play distinct roles. The pair $(\cat M,R)$ supplies the
\emph{syntax}: lenses in $\cat M$ carry the wiring, and the monad $R\times\blank$
adds a potential to the backward pass, giving the coKleisli category
$\Lcokl{\cat M}{R}$. The functor $J$ supplies \emph{parameters}, acting by
$a\cdot\lensob c=\binom{\inpt c}{J(a)\otimes\outp c}$. The functor $\Fun{c}$ is the
\emph{bridge} to polynomials: strong monoidality is what turns a parallel product
of interfaces into a $\otimes$-product of polynomials, and it carries the receptor
$R$ to a commutative monoid $\Fun{c}(R)$ in $\poly$. The potential algebra
$(z,\alpha)$ is the recipient into which a potential differentiates. We do
\emph{not} require $\cat M$ to be monoidal closed: Moore internalization is carried
out in $\poly$, after $\Fun{c}$, so an internal hom is needed only in the (fixed,
closed) target.

\section{The syntax operad of a setup}\label{sec.rwd_syntax}
%----------------------------------------------------------------------------

\begin{definition}[$\rwd_D$]\label{def.rwd}
Let $D$ be a rewiring setup. The symmetric monoidal category
$\para{\cat V}{\Lcokl{\cat M}{R}}$ has, as its underlying operad, the
\emph{rewiring-diagram operad} $\rwd_D$ of $D$.
\end{definition}

\section{Moore internalization and the data functor}\label{sec.data_functor}
%----------------------------------------------------------------------------

% RELOCATE HERE (verbatim): sec.abstract_representation -- Moore internalization
% Theta_z (thm.Theta_T_alpha) is already stated for an arbitrary SMCC, monoidal
% monad T, and T-monoid z; here it is applied at the (fixed, closed) target Poly.

\begin{theorem}[Data functor]\label{thm.data_functor}
Every rewiring setup $D$ yields a normal lax symmetric monoidal functor
\[
  \Phi'\colon\rwd_D\longrightarrow\para{\Fun{c}}{\poly},
\]
where $\cat V$ acts on $\poly$ by $a\cdot q\coloneqq \Fun{c}(Ja)\otimes q$. On
objects, $\Phi'\lensob{M}=\Fun{c}(\outp{M})\otimes\ihom{\Fun{c}(\inpt{M}),z}$.
\end{theorem}

\begin{proof}
% TODO: relocate the proof of lem.potlens_to_para_poly verbatim, cot -> c, mfd -> M.
Compose the two normal lax monoidal functors
\[
  \para{\cat V}{\Lcokl{\cat M}{R}}\To{\para{\cat V}{\Lens{\Fun{c}}}}
  \para{\cat V}{\Lcokl{\poly}{\Fun{c}(R)}}\To{\para{\cat V}{\Theta_z}}
  \para{\Fun{c}}{\poly}
\]
and pass to underlying operads. The first leg is the $\Fun{c}$-level analogue of
\cref{lem.cot_lifts_lens_potential} (strong monoidality of $\Fun{c}$, together with
the monad morphism $\Fun{c}(R\times\blank)\Rightarrow\Fun{c}(R)\otimes\Fun{c}(\blank)$
supplied by \cref{lemma.monoid_to_monad}); the second is
\cref{cor.parameterized_representation} applied to the potential algebra $(z,\alpha)$.
\end{proof}

Let $D$ be a rewiring setup and $\intg$ a $(\Fun{c}\circ J)$-integrator. There is an
operad functor
\[
  \Phi_\intg\colon\rwd_D\longrightarrow\org,
\]
namely the composite of the data functor $\Phi'$ (\cref{thm.data_functor}) with
the integrator semantics $\Psisem{\intg}$ (\cref{prop.integrator_to_org}).
\end{theorem}

\begin{proof}

\end{proof}

Thus the construction takes in data and returns a category over the fixed semantic
anchor $\org$.

\begin{remark}[Functoriality in the data]\label{rmk.rwd_functor}
Write $\Data$ for the category of rewiring setups-with-integrator $(D,\intg)$
[\,morphisms: TODO\,]. The construction above is the object part of a functor
\[
  \rwd\colon\Data\longrightarrow\SMC_{/\org},
  \qquad (D,\intg)\longmapsto\bigl(\rwd_D,\ \Phi_\intg\bigr),
\]
landing in the (normal-lax, pseudo) slice of symmetric monoidal categories over
$\org$.
\end{remark}

\begin{remark}[The integrator design space]\label{rmk.integrator_design}
For a fixed setup the integrators of interest---configuration, phase, dissipative,
look-ahead---form a design space, and one might ask whether $\intg\mapsto\Phi_\intg$
is functorial in the integrator, i.e.\ whether a suitable notion of morphism of
integrators makes it a functor $\mathbf{Int}_D\to[\rwd_D,\org]$. We do not pursue
this. The natural relations among our integrators are not maps of integrators in any
evident sense: configuration ($\Fun S=\absval\blank$) and phase
($\Fun S=\absval{T^*\blank}$) have different state spaces and step along different
data---the phase update uses the \emph{stored} momentum, the configuration update the
\emph{incoming} covector---so the projection $T^*V\to V$ does not intertwine their
updates; and the friction parameter of the dissipative family is not natural.
Functoriality in the \emph{setup} (\cref{rmk.rwd_functor}) is the better-supported
direction.
\end{remark}

\section{Recovery: the smooth instance}\label{sec.recovery_DRAFT}
%----------------------------------------------------------------------------

\begin{proposition}[The smooth setup]\label{prop.smooth_setup}
The tuple
\[
  D_{\mathrm{sm}}\coloneqq(\mfd,\ \rvect,\ \inc,\ \rr,\ \cot,\ \yon,\ \potd)
\]
is a rewiring setup (\cref{def.rewiring_setup}).
\end{proposition}

\begin{proof}
We check the five clauses of \cref{def.rewiring_setup}.
\begin{enumerate}
  \item $(\mfd,\rr^0,\times)$ is cartesian.
  \item $(\rvect,0,\oplus)$ is symmetric monoidal (\cref{prop.rvect_monoidal}) and
    $\inc\colon\rvect\to\mfd$ is strong symmetric monoidal
    (\cref{prop.rvect_smc_action}).
  \item $(\rr,0,+)$ is a commutative monoid---indeed a Lie group---object in $\mfd$.
  \item $\cot\colon(\mfd,\rr^0,\times)\to(\poly,\yon,\otimes)$ is strong symmetric
    monoidal (\cref{prop.cot_monoidal}), carrying $\rr$ to the commutative monoid
    $\cotof{\rr}$ in $\poly$ (\cref{prop.cot_hopf}).
  \item $\yon$ is a $\otimes$-monoid, and $\potd\colon\cotof{\rr}\to\yon$ is a
    $(\cotof{\rr}\otimes\blank)$-monoid structure on it---the choice $r=+1$ of
    \eqref{eqn.d_potential}, by \cref{lem.alpha_constant}.
    \qedhere
\end{enumerate}
\end{proof}

\begin{theorem}[Recovery]\label{thm.recovery}
For the smooth setup $D_{\mathrm{sm}}$ of \cref{prop.smooth_setup}:
\begin{enumerate}
  \item $\rwd_{D_{\mathrm{sm}}}=\srwd$ (\cref{def.potlens}), and the data functor of
    \cref{thm.data_functor} is the $\Phi'$ of \cref{lem.potlens_to_para_poly};
  \item the configuration and phase integrators $\intg_\theta$
    \eqref{eqn.conf_integrator} and $\intg_\beta$ \eqref{eqn.phase_integrator} are
    $(\cot\circ\inc)$-integrators, and the dynamics functors they induce via
    \cref{thm.dynamics_functor} are
    \[
      \Phi_{\intg_\theta}=\Phiconf
      \qqand
      \Phi_{\intg_\beta}=\Phiphase
    \]
    of \eqref{eqn.our_phis}.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) By \cref{def.rwd}, $\rwd_{D_{\mathrm{sm}}}$ is the underlying operad of
$\para{\rvect}{\Lcokl{\mfd}{\rr}}=\para{\rvect}{\plmfd}$, which is $\srwd$
(\cref{def.potlens}). With $\Fun{c}\coloneqq\cot$ and $(z,\alpha)\coloneqq(\yon,\potd)$,
the two legs of \cref{thm.data_functor} are $\Lens\cot$
(\cref{lem.cot_lifts_lens_potential}) and $\Theta_\potd$
(\cref{lem.Theta_poly_potential}), so the data functor is the $\Phi'$ of
\cref{lem.potlens_to_para_poly}.

(2) Both $\intg_\theta=(\absval\blank,\theta)$ and
$\intg_\beta=(\absval{T^*\blank},\upd_\beta)$ are $\cot$-integrators
(\cref{def.integrator}), as established at \eqref{eqn.conf_integrator} and in
\cref{sec.phase_integrators}. The induced functor
$\Phi_{\intg}=\Psisem{\intg}\circ\Phi'$ of \cref{thm.dynamics_functor} is that of
\cref{thm.functor}; the two named cases are \eqref{eqn.our_phis}.
\qedhere
\end{proof}

From here on we write $\srwd$ for $\rwd_{D_{\mathrm{sm}}}$ and call its morphisms
\emph{smooth rewiring diagrams}; the word ``smooth'' refers to this instance.
